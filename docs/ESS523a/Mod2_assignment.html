Find the average flipper length for each species. Which species has the largest flippers? (2 pts.)

penguins %>% 
  select(species, flipper_length_mm) %>% #select relevant columns
  group_by(species) %>% # group by species
  summarise(flipper_length_avg = mean(flipper_length_mm, na.rm = TRUE)) %>% #calculate mean and make new df
  slice_max(flipper_length_avg, n=1) %>%  #take largest value
print()
## # A tibble: 1 × 2
